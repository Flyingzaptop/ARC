cmake_minimum_required(VERSION 3.24)
project(arc_cpu_backend LANGUAGES C CXX ASM_MASM)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "ARC CPU DynamoRIO backend requires Windows x64")
endif()
if(NOT DynamoRIO_DIR)
  message(FATAL_ERROR "Pass -DDynamoRIO_DIR=<pinned DynamoRIO>/cmake")
endif()
find_package(DynamoRIO 11.3 EXACT CONFIG REQUIRED)
add_library(arc_cpu_client SHARED client.cpp)
target_compile_features(arc_cpu_client PRIVATE cxx_std_20)
target_compile_definitions(arc_cpu_client PRIVATE NOMINMAX)
target_include_directories(arc_cpu_client PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../../include")
configure_DynamoRIO_client(arc_cpu_client)
use_DynamoRIO_extension(arc_cpu_client drmgr)
set_target_properties(arc_cpu_client PROPERTIES OUTPUT_NAME "arc_cpu_client")

add_executable(cpu_native_fixture "${CMAKE_CURRENT_LIST_DIR}/../../../tests/cpu_native_fixture.cpp"
                                  "${CMAKE_CURRENT_LIST_DIR}/../../../tests/cpu_native_regions.asm")
target_compile_features(cpu_native_fixture PRIVATE cxx_std_20)
